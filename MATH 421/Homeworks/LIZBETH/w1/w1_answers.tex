\section*{Section A: Simplify (Problems 1--4)}
Recall the \textbf{Product of Powers Rule}: $a^m \cdot a^n = a^{m+n}$, and the \textbf{Power of a Power Rule}: $(a^m)^n = a^{mn}$.

\begin{enumerate}
\item
Original expression: \quad $x^3 \cdot x^5$

\begin{align*}
x^3 \cdot x^5 &= x^{3+5} && \text{(Product of powers: add exponents)} \\
&= x^8 \\
\boxed{x^8}
\end{align*}

\item
Original expression: \quad $(y^4)^3$

\begin{align*}
(y^4)^3 &= y^{4 \cdot 3} && \text{(Power of a power: multiply exponents)} \\
&= y^{12} \\
\boxed{y^{12}}
\end{align*}

\item
Original expression: \quad $a^7 \cdot a^{-2}$

\begin{align*}
a^7 \cdot a^{-2} &= a^{7 + (-2)} && \text{(Product of powers)} \\
&= a^5 \\
\boxed{a^5}
\end{align*}

\item
Original expression: \quad $(b^2)^0$

\begin{align*}
(b^2)^0 &= b^{2 \cdot 0} && \text{(Power of a power, and any base to zero is 1)} \\
&= b^0 \\
&= 1 \\
\boxed{1}
\end{align*}
\end{enumerate}

\section*{Section B: Evaluate (Problems 5--8)} 
Here we expand powers and multiply or divide numbers step-by-step.

\begin{enumerate}
\setcounter{enumi}{4}

\item
Original expression: \quad $2^3 \cdot 3^2$

\begin{align*}
2^3 \cdot 3^2 &= (2 \cdot 2 \cdot 2) \cdot (3 \cdot 3) \\
&= 8 \cdot 9 \\
&= 72 \\
\boxed{72}
\end{align*}

\item
Original expression: \quad $\frac{5^4}{5^2}$

\begin{align*}
\frac{5^4}{5^2} &= 5^{4 - 2} && \text{(Quotient of powers: subtract exponents)} \\
&= 5^2 \\
&= 25 \\
\boxed{25}
\end{align*}

\item
Original expression: \quad $(4^2)^3$

\begin{align*}
(4^2)^3 &= 4^{2 \cdot 3} && \text{(Power of a power)} \\
&= 4^6 \\
&= 4 \cdot 4 \cdot 4 \cdot 4 \cdot 4 \cdot 4 \\
&= 4096 \\
\boxed{4096}
\end{align*}

\item
Original expression: \quad $\frac{10^5}{2 \cdot 5^3}$

\begin{align*}
\frac{10^5}{2 \cdot 5^3} &= \frac{(2 \cdot 5)^5}{2 \cdot 5^3} && \text{(Write }10=2 \cdot 5) \\
&= \frac{2^5 \cdot 5^5}{2 \cdot 5^3} \\
&= \frac{2^5}{2^1} \cdot \frac{5^5}{5^3} && \text{(Separate powers)} \\
&= 2^{5-1} \cdot 5^{5-3} && \text{(Quotient of powers)} \\
&= 2^4 \cdot 5^2 \\
&= 16 \cdot 25 \\
&= 400 \\
\boxed{400}
\end{align*}
\end{enumerate}

\section*{Section C: Simplify Completely (Problems 9--13)} 
We will use exponent rules including product, quotient, power of a power, and negative exponent rules. Recall the \textbf{Negative Exponent Rule}: $a^{-n} = \frac{1}{a^n}$.

\begin{enumerate}
\setcounter{enumi}{8}

\item
Original expression: \quad $\frac{x^7}{x^3}$

\begin{align*}
\frac{x^7}{x^3} &= x^{7-3} && \text{(Quotient of powers)} \\
&= x^4 \\
\boxed{x^4}
\end{align*}

\item
Original expression: \quad $(m^3 n^2)^4$

\begin{align*}
(m^3 n^2)^4 &= m^{3 \cdot 4} n^{2 \cdot 4} && \text{(Power of a product)} \\
&= m^{12} n^{8} \\
\boxed{m^{12} n^{8}}
\end{align*}

\item
Original expression: \quad $\frac{a^5 b^{-2}}{a^{-3} b^4}$

\begin{align*}
\frac{a^5 b^{-2}}{a^{-3} b^4} &= a^{5 - (-3)} b^{-2 - 4} && \text{(Quotient of powers)} \\
&= a^{5 + 3} b^{-6} \\
&= a^{8} b^{-6} \\
\boxed{a^{8} b^{-6}}
\end{align*}

\item
Original expression: \quad $(2x^3 y^{-1})^2$

\begin{align*}
(2x^3 y^{-1})^2 &= 2^2 \cdot (x^3)^2 \cdot (y^{-1})^2 \\
&= 4 \cdot x^{3 \cdot 2} \cdot y^{-2} \\
&= 4 x^{6} y^{-2} \\
\boxed{4 x^{6} y^{-2}}
\end{align*}

\item
Original expression: \quad $\frac{(p^2 q^{-3})^3}{p^{-4} q^2}$

\begin{align*}
\frac{(p^2 q^{-3})^3}{p^{-4} q^2} &= \frac{p^{2 \cdot 3} q^{-3 \cdot 3}}{p^{-4} q^2} \\
&= \frac{p^{6} q^{-9}}{p^{-4} q^{2}} \\
&= p^{6 - (-4)} q^{-9 - 2} && \text{(Quotient of powers)} \\
&= p^{10} q^{-11} \\
\boxed{p^{10} q^{-11}}
\end{align*}
\end{enumerate}

\section*{Section D: Simplify Completely (Problems 14--17)} 
Again, apply all exponent rules carefully.

\begin{enumerate}
\setcounter{enumi}{13}

\item
Original expression: \quad $\left(\frac{x^2 y^{-3}}{x^{-1} y^4}\right)^2$

\begin{align*}
\left(\frac{x^2 y^{-3}}{x^{-1} y^4}\right)^2 &= \left(x^{2 - (-1)} y^{-3 - 4}\right)^2 && \text{(Quotient of powers)} \\
&= \left(x^{3} y^{-7}\right)^2 \\
&= x^{3 \cdot 2} y^{-7 \cdot 2} && \text{(Power of a power)} \\
&= x^{6} y^{-14} \\
\boxed{x^{6} y^{-14}}
\end{align*}

\item
Original expression: \quad $\frac{(2a^{-1} b^3)^2}{4 a^2 b^{-2}}$

\begin{align*}
\frac{(2a^{-1} b^3)^2}{4 a^2 b^{-2}} &= \frac{2^2 \cdot a^{-1 \cdot 2} \cdot b^{3 \cdot 2}}{4 a^{2} b^{-2}} \\
&= \frac{4 a^{-2} b^{6}}{4 a^{2} b^{-2}} \\
&= \frac{4}{4} \cdot a^{-2 - 2} \cdot b^{6 - (-2)} && \text{(Quotient of powers)} \\
&= a^{-4} b^{8} \\
\boxed{a^{-4} b^{8}}
\end{align*}

\item
Original expression: \quad $(x^0 y^5)^{-3}$

\begin{align*}
(x^0 y^5)^{-3} &= x^{0 \cdot (-3)} y^{5 \cdot (-3)} && \text{(Power of a product)} \\
&= x^0 y^{-15} \\
&= y^{-15} \\
\boxed{y^{-15}}
\end{align*}

\item
Original expression: \quad $x^{-2} \cdot \frac{1}{x^{-5}}$

\begin{align*}
x^{-2} \cdot \frac{1}{x^{-5}} &= x^{-2} \cdot x^{5} && \text{(Negative exponent rule)} \\
&= x^{-2 + 5} && \text{(Product of powers)} \\
&= x^{3} \\
\boxed{x^{3}}
\end{align*}
\end{enumerate}

\section*{Section E: Simplify with Positive Exponents (Problems 18--22)} 
Write final answers with only positive exponents.

\begin{enumerate}
\setcounter{enumi}{17}

\item
Original expression: \quad $\frac{x^{-3} y^4}{x^2 y^{-1}}$

\begin{align*}
\frac{x^{-3} y^4}{x^2 y^{-1}} &= x^{-3 - 2} y^{4 - (-1)} && \text{(Quotient of powers)} \\
&= x^{-5} y^{5} \\
&= \frac{y^{5}}{x^{5}} && \text{(Negative exponent rule)} \\
\boxed{\frac{y^{5}}{x^{5}}}
\end{align*}

\item
Original expression: \quad $(a^{-2} b^3)^2$

\begin{align*}
(a^{-2} b^3)^2 &= a^{-2 \cdot 2} b^{3 \cdot 2} && \text{(Power of a product)} \\
&= a^{-4} b^{6} \\
&= \frac{b^{6}}{a^{4}} && \text{(Negative exponent rule)} \\
\boxed{\frac{b^{6}}{a^{4}}}
\end{align*}

\item
Original expression: \quad $\frac{(x^3 y^{-2})^{-2}}{x^{-4} y^3}$

\begin{align*}
\frac{(x^3 y^{-2})^{-2}}{x^{-4} y^3} &= \frac{x^{3 \cdot (-2)} y^{-2 \cdot (-2)}}{x^{-4} y^{3}} \\
&= \frac{x^{-6} y^{4}}{x^{-4} y^{3}} \\
&= x^{-6 - (-4)} y^{4 - 3} && \text{(Quotient of powers)} \\
&= x^{-2} y^{1} \\
&= \frac{y}{x^{2}} && \text{(Negative exponent rule)} \\
\boxed{\frac{y}{x^{2}}}
\end{align*}

\item
Original expression: \quad $\left(\frac{a^{-1} b^2}{a^3 b^{-4}}\right)^{-2}$

\begin{align*}
\left(\frac{a^{-1} b^2}{a^3 b^{-4}}\right)^{-2} &= \left(a^{-1 - 3} b^{2 - (-4)}\right)^{-2} && \text{(Quotient of powers)} \\
&= \left(a^{-4} b^{6}\right)^{-2} \\
&= a^{-4 \cdot (-2)} b^{6 \cdot (-2)} && \text{(Power of a power)} \\
&= a^{8} b^{-12} \\
&= \frac{a^{8}}{b^{12}} && \text{(Negative exponent rule)} \\
\boxed{\frac{a^{8}}{b^{12}}}
\end{align*}

\item
Original expression: \quad $\frac{(x^{-2} y^3)^3}{(x y^{-1})^4}$

\begin{align*}
\frac{(x^{-2} y^3)^3}{(x y^{-1})^4} &= \frac{x^{-2 \cdot 3} y^{3 \cdot 3}}{x^{4} y^{-1 \cdot 4}} \\
&= \frac{x^{-6} y^{9}}{x^{4} y^{-4}} \\
&= x^{-6 - 4} y^{9 - (-4)} && \text{(Quotient of powers)} \\
&= x^{-10} y^{13} \\
&= \frac{y^{13}}{x^{10}} && \text{(Negative exponent rule)} \\
\boxed{\frac{y^{13}}{x^{10}}}
\end{align*}
\end{enumerate}

\section*{Section F: Word Problems (Problems 23--25)} 
Show unit conversions and exponent arithmetic carefully. Final answers for problem 25 must be in standard form.

\begin{enumerate}
\setcounter{enumi}{22}

\item
Original problem: A cube has side length $5 \times 10^2$ cm. Find its volume in cubic meters.

\begin{align*}
\text{Side length} &= 5 \times 10^2 \text{ cm} \\
&= 5 \times 10^2 \times \frac{1 \text{ m}}{100 \text{ cm}} && \text{(Convert cm to m)} \\
&= 5 \times 10^2 \times 10^{-2} \text{ m} \\
&= 5 \times 10^{2 - 2} \text{ m} = 5 \times 10^{0} \text{ m} = 5 \text{ m} \\
\text{Volume} &= (\text{side length})^3 = (5 \text{ m})^3 = 5^3 \text{ m}^3 = 125 \text{ m}^3 \\
\boxed{125 \text{ m}^3}
\end{align*}

\item
Original problem: The population of a city is $3 \times 10^6$. If it doubles every 5 years, what is the population after 10 years?

\begin{align*}
\text{After 5 years:} &\quad 3 \times 10^6 \times 2 = 6 \times 10^6 \\
\text{After 10 years:} &\quad 6 \times 10^6 \times 2 = 12 \times 10^6 = 1.2 \times 10^7 \\
\boxed{1.2 \times 10^{7}}
\end{align*}

\item
Original problem: Express $(4 \times 10^{-3})^3$ in standard form.

\begin{align*}
(4 \times 10^{-3})^3 &= 4^3 \times (10^{-3})^3 && \text{(Power of a product)} \\
&= 64 \times 10^{-9} \\
&= 6.4 \times 10^{1} \times 10^{-9} && \text{(Rewrite }64 = 6.4 \times 10^1) \\
&= 6.4 \times 10^{1 - 9} \\
&= 6.4 \times 10^{-8} \\
\boxed{6.4 \times 10^{-8}}
\end{align*}
\end{enumerate}
