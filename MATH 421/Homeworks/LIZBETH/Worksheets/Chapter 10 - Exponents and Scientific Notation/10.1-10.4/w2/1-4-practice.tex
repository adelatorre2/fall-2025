% =========================================================
% 10.1 Writing products using exponents
% =========================================================
\textbf{A. Write the product using exponents (Section 10.1).}

\begin{enumerate}[label=\arabic*.]
\item $(-6)(-6)(-6)$
\item $4\cdot 4\cdot 4\cdot k\cdot k$
\item $y\cdot y\cdot y\cdot y\cdot z$
\item $m\cdot m\cdot n\cdot n\cdot n\cdot n$
\end{enumerate}

\hrule

% =========================================================
% 10.1 Evaluating powers
% =========================================================
\textbf{B. Evaluate (Section 10.1).}

\begin{enumerate}[label=\arabic*., resume]
\item $2^7$
\item $(-4)^3$
\item $(-1)^8$
\item $\left(\dfrac{5}{6}\right)^2$
\end{enumerate}

\vspace{0.4em}
\hrule
\vspace{0.8em}

% =========================================================
% 10.2 Exponent rules
% =========================================================
\textbf{C. Simplify. Write your answer as a single power (Section 10.2).}

\begin{enumerate}[label=\arabic*., resume]
\item $10^4 \cdot 10^9$
\item $\dfrac{8^{12}}{8^7}$
\item $(q^6)^2$
\item $(4a^3b^2)^2$
\item $\left(\dfrac{2}{5}\right)^7 \left(\dfrac{2}{5}\right)^3$
\end{enumerate}

\vspace{0.4em}
\hrule
\vspace{0.8em}

% =========================================================
% 10.3 More exponent properties
% =========================================================
\textbf{D. Simplify (Section 10.3).}

\begin{enumerate}[label=\arabic*., resume]
\item $\dfrac{9^5 \cdot 9^2}{9^4}$
\item $\dfrac{w^{14} \cdot w^3}{w^9}$
\item $\dfrac{p^{18}}{p^6 \cdot p^5}$
\item $\left(\dfrac{5}{9}x\right)^2$
\end{enumerate}

\vspace{0.4em}
\hrule

\textbf{E. Rewrite using only positive exponents (Section 10.4).}

\begin{enumerate}[label=\arabic*., resume]
\item $7n^{-3}$
\item $\dfrac{20y^4}{5y^{-6}}$
\item $\dfrac{r^{-5}s^3}{r^2 s^{-4}}$

\end{enumerate}

% =========================================================
% Applications
% =========================================================
\textbf{F. Applications (mixed 10.1--10.4).}

\begin{enumerate}[label=\arabic*., resume]
\item A virus is $4200$ micrometers long. If $1$ micrometer $=10^{-6}$ meters, what is the length in meters?
\item A number is multiplied by $10^6$ and then divided by $10^4$. Overall, what power of 10 is the number multiplied by?
\item A printer produces $5\times10^4$ pages per day. How many pages are printed in $4\times10^1$ days?  
(Write your answer in standard form.)
\end{enumerate}
